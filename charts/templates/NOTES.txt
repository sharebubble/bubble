1. Accessing the application:
{{- if not .Values.frontend.enabled }}
   - The frontend component is disabled in this release.
{{- else if .Values.frontend.ingress.enabled }}
   - Ingress: {{ include "bubble.frontend.fullname" . }} (namespace {{ .Release.Namespace }})
{{- if .Values.frontend.ingress.tls }}
   - URL(s):
{{- range .Values.frontend.ingress.hosts }}
     * https://{{ .host }}
{{- end }}
{{- else }}
   - URL(s):
{{- range .Values.frontend.ingress.hosts }}
     * http://{{ .host }}
{{- end }}
{{- end }}
{{- else if .Values.frontend.httpRoute.enabled }}
   - HTTPRoute: {{ include "bubble.frontend.fullname" . }} (namespace {{ .Release.Namespace }})
{{- if .Values.frontend.httpRoute.parentRefs }}
   - Attached Gateway reference(s):
{{- range .Values.frontend.httpRoute.parentRefs }}
     * name={{ .name }}{{- if .namespace }}, namespace={{ .namespace }}{{- end }}{{- if .sectionName }}, section={{ .sectionName }}{{- end }}
{{- end }}
{{- else if .Values.frontend.gateway.enabled }}
   - Attached Gateway: {{ include "bubble.frontend.gateway.fullname" . }} (namespace {{ .Release.Namespace }})
     kubectl get gateway {{ include "bubble.frontend.gateway.fullname" . }} -n {{ .Release.Namespace }}
{{- end }}
{{- if .Values.frontend.httpRoute.hostnames }}
   - Hostnames:
{{- range .Values.frontend.httpRoute.hostnames }}
     * {{ . }}
{{- end }}
{{- else }}
   - Hostnames follow the listener configuration on the parent Gateway.
{{- end }}
{{- else if contains "NodePort" .Values.frontend.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "bubble.frontend.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.frontend.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "bubble.frontend.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "bubble.frontend.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.frontend.service.port }}
{{- else if contains "ClusterIP" .Values.frontend.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "bubble.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=frontend" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

3. Important notes:
   - Make sure to set proper secrets in values.yaml or use external secret management
{{- if .Values.frontend.ingress.enabled }}
   - For production, configure TLS in frontend.ingress.tls and set a real hostname in frontend.ingress.hosts
{{- else if .Values.frontend.httpRoute.enabled }}
   - For production, terminate TLS on the Gateway listener and set proper hostnames
{{- end }}
   - Consider using external PostgreSQL and Redis for production workloads
